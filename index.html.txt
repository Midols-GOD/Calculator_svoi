        function setLanguage(lang) {
            currentLang = lang;
            document.documentElement.lang = lang === 'ua' ? 'uk' : 'ru';
            document.getElementById('btnLangRu').className = lang === 'ru'
                ? 'px-3 py-1.5 rounded-lg bg-orange-600 text-white transition cursor-pointer'
                : 'px-3 py-1.5 rounded-lg text-slate-300 hover:text-white transition cursor-pointer';
            document.getElementById('btnLangUa').className = lang === 'ua'
                ? 'px-3 py-1.5 rounded-lg bg-orange-600 text-white transition cursor-pointer'
                : 'px-3 py-1.5 rounded-lg text-slate-300 hover:text-white transition cursor-pointer';

            document.querySelectorAll('[data-i18n]').forEach(el => {
                const key = el.getAttribute('data-i18n');
                if (I18N[lang][key]) {
                    el.textContent = I18N[lang][key];
                }
            });

            renderOptions();
            calculate();
        }

        function selectSurface(surf) {
            state.surface = surf;
            const cfg = SURFACE_CONFIG[surf];
            if (!cfg.bases.includes(state.base)) {
                state.base = cfg.bases[0];
                document.getElementById('inpLayerMm').value = DEFAULT_LAYERS_MM[state.base] || 10;
            }
            if (!cfg.goals.includes(state.goal)) {
                state.goal = cfg.goals[0];
            }
            updateDimensionVisibility();
            renderOptions();
            calculate();
        }

        function selectBase(base) {
            state.base = base;
            document.getElementById('inpLayerMm').value = DEFAULT_LAYERS_MM[base] || 10;
            renderOptions();
            calculate();
        }

        function selectGoal(goal) {
            state.goal = goal;
            renderOptions();
            calculate();
        }

        function updateDimensionVisibility() {
            const roomBox = document.getElementById('roomDimensionsBox');
            const heightWrap = document.getElementById('heightFieldWrap');
            const deductBox = document.getElementById('openingsDeductionBox');
            const openBox = document.getElementById('openingDimensionsBox');

            if (state.surface === 'opening') {
                roomBox.classList.add('hidden');
                deductBox.classList.add('hidden');
                openBox.classList.remove('hidden');
            } else {
                roomBox.classList.remove('hidden');
                openBox.classList.add('hidden');
                if (state.surface === 'walls') {
                    heightWrap.classList.remove('hidden');
                    deductBox.classList.remove('hidden');
                } else {
                    heightWrap.classList.add('hidden');
                    deductBox.classList.add('hidden');
                }
            }
        }

        function getNetArea() {
            if (state.surface === 'opening') {
                const h = Math.max(0.1, parseFloat(document.getElementById('inpOpenHeight').value) || 0);
                const w = Math.max(0.1, parseFloat(document.getElementById('inpOpenWidth').value) || 0);
                const dCm = Math.max(1, parseFloat(document.getElementById('inpOpenDepth').value) || 0);
                const cnt = Math.max(1, parseInt(document.getElementById('inpOpenCount').value, 10) || 1);
                const perimeter = (h * 2) + w;
                return perimeter * (dCm / 100) * cnt;
            }

            const l = Math.max(0.1, parseFloat(document.getElementById('inpLength').value) || 0);
            const w = Math.max(0.1, parseFloat(document.getElementById('inpWidth').value) || 0);

            if (state.surface === 'ceiling' || state.surface === 'floor') {
                return l * w;
            }

            const h = Math.max(0.1, parseFloat(document.getElementById('inpHeight').value) || 0);
            const doors = Math.max(0, parseInt(document.getElementById('inpDoorsCount').value, 10) || 0);
            const windows = Math.max(0, parseInt(document.getElementById('inpWindowsCount').value, 10) || 0);

            const gross = 2 * (l + w) * h;
            const deduct = (doors * 1.8) + (windows * 1.96);
            return Math.max(0.5, gross - deduct);
        }

        function buildMaterials(areaWithReserve, layerMm) {
            const isUa = currentLang === 'ua';

            const steps = [];

                const packs = Math.max(1, Math.ceil(rawQty / packSize));

                        isUa ? 'Шумо- та теплоізоляція основи' : 'Шумо- и теплоизоляция основания',
                        areaWithReserve,
                        'м²',
                        5,
                        isUa ? 'упаковок (5 м²)' : 'упаковок (5 м²)'
                    );
                    addMat(
                        'laminate',
                        isUa ? 'Ламінат / паркетна дошка' : 'Ламинат / паркетная доска',

                        'mesh',
                        isUa ? 'Сітка штукатурна скловолоконна (5×5 мм)' : 'Сетка штукатурная стекловолоконная (5×5 мм)',

                        areaWithReserve * 1.1,
                        'м²',
                        50,
                        isUa ? 'рулонів (50 м²)' : 'рулонов (50 м²)'
                    );
                }

                addStep(isUa ? '3. Чорнова штукатурка' : '3. Черновая штукатурка');
                const plasterRate = (state.goal === 'tile' || state.base === 'stone') ? 1.4 : 0.95;
                const plasterName = (state.goal === 'tile' || state.base === 'stone')
                    ? (isUa ? 'Штукатурка цементно-піщана / фасадна' : 'Штукатурка цементно-песчаная / фасадная')

                    plasterName,
                    isUa ? `Шар ${layerMm} мм (витрата ~${plasterRate} кг/м² на 1 мм)` : `Слой ${layerMm} мм (расход ~${plasterRate} кг/м² на 1 мм)`,

            }

            if (state.surface === 'opening') {
                const h = Math.max(0.1, parseFloat(document.getElementById('inpOpenHeight').value) || 0);

                    areaWithReserve * 1.2,
                    'м.п.',
                    45,

            }

            const puttyLayers = state.goal === 'paint' ? 2 : 1;

                    : (isUa ? 'Під шпалери достатньо 1–2 тонких шарів' : 'Под обои достаточно 1–2 тонких слоёв'),

            if (state.goal === 'wallpaper') {

        function calculate() {
            const netArea = getNetArea();
            const reservePct = parseFloat(document.getElementById('inpReserve').value) || 0;
            const areaWithReserve = netArea * (1 + reservePct / 100);
            const layerMm = Math.max(1, parseFloat(document.getElementById('inpLayerMm').value) || 10);

            document.getElementById('resNetArea').textContent = `${netArea.toFixed(2)} м²`;

            const { materials, steps } = buildMaterials(areaWithReserve, layerMm);

            const stepsBox = document.getElementById('workStepsPills');
            stepsBox.innerHTML = steps.map(s => `
                <span class="px-2.5 py-1 rounded-lg bg-slate-800 border border-slate-700 text-xs font-semibold text-slate-200">${s}</span>
            `).join('');

            const listBox = document.getElementById('materialsResultList');
            const ph = I18N[currentLang].pricePlaceholder;

            listBox.innerHTML = materials.map(m => {
                const savedPrice = unitPrices[m.id] || '';
                return `
                    <div class="p-3.5 rounded-xl bg-slate-800/75 border border-slate-700/90">
                        <div class="flex justify-between items-start gap-3">
                            <div>
                                <div class="font-bold text-sm text-white leading-snug">${m.name}</div>
                                <div class="text-[11px] text-slate-400 mt-0.5">${m.desc}</div>
                            </div>
                            <div class="text-right flex-shrink-0">
                                <div class="text-base font-black text-orange-400">${m.packs} ${m.packUnitName}</div>
                                <div class="text-[11px] text-slate-400">≈ ${m.rawQty} ${m.rawUnit}</div>
                            </div>
                        </div>
                        <div class="mt-2.5 pt-2.5 border-t border-slate-700/70 flex items-center justify-between gap-2">
                            <span class="text-[11px] text-slate-400">${ph}:</span>
                            <div class="flex items-center gap-1.5">
                                <input type="number" min="0" step="1" value="${savedPrice}" placeholder="0"
                                    oninput="updatePrice('${m.id}', this.value)"
                                    class="w-24 rounded-lg bg-slate-900 border border-slate-600 px-2.5 py-1 text-right text-xs font-bold text-amber-400 focus:border-orange-500 focus:outline-none">
                                <span class="text-xs text-slate-400 font-bold">₴</span>
                            </div>
                        </div>
                    </div>
                `;
            }).join('');

            recalcTotalMoney(materials);
        }

        function updatePrice(matId, val) {
            unitPrices[matId] = parseFloat(val) || 0;
            const netArea = getNetArea();
            const reservePct = parseFloat(document.getElementById('inpReserve').value) || 0;
            const areaWithReserve = netArea * (1 + reservePct / 100);
            const layerMm = Math.max(1, parseFloat(document.getElementById('inpLayerMm').value) || 10);
            const { materials } = buildMaterials(areaWithReserve, layerMm);
            recalcTotalMoney(materials);
        }

        function recalcTotalMoney(materials) {
            let total = 0;
            materials.forEach(m => {
                const p = unitPrices[m.id] || 0;
                total += p * m.packs;
            });
            document.getElementById('resTotalCost').textContent = `${Math.round(total).toLocaleString()} ₴`;
        }

        function copyEstimate() {
            const t = I18N[currentLang];
            const netArea = getNetArea();
            const reservePct = parseFloat(document.getElementById('inpReserve').value) || 0;
            const areaWithReserve = netArea * (1 + reservePct / 100);
            const layerMm = Math.max(1, parseFloat(document.getElementById('inpLayerMm').value) || 10);
            const { materials } = buildMaterials(areaWithReserve, layerMm);

            let lines = [
                `📋 ${t.brandName} — ${t.surfaces[state.surface].title}`,
                `🧱 ${t.bases[state.base].title}`,
                `🎯 ${t.goals[state.goal].title}`,
                `📐 ${t.resAreaLbl}: ${netArea.toFixed(2)} м² (+${reservePct}%)`,
                `-------------------------`
            ];

            let totalMoney = 0;
            materials.forEach((m, idx) => {
                const p = unitPrices[m.id] || 0;
                const sumStr = p > 0 ? ` — ${Math.round(p * m.packs)} ₴` : '';
                totalMoney += p * m.packs;
                lines.push(`${idx + 1}. ${m.name}: ${m.packs} ${m.packUnitName} (~${m.rawQty} ${m.rawUnit})${sumStr}`);
            });

            if (totalMoney > 0) {
                lines.push(`-------------------------`);
                lines.push(`💰 ${t.resTotalMoneyLbl} ${Math.round(totalMoney).toLocaleString()} ₴`);
            }

            const textToCopy = lines.join('
');
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(textToCopy);
            } else {
                const tmp = document.createElement('textarea');
                tmp.value = textToCopy;
                document.body.appendChild(tmp);
                tmp.select();
                document.execCommand('copy');
                document.body.removeChild(tmp);
            }

            const btnText = document.getElementById('copyBtnText');
            btnText.textContent = t.copiedBtn;
            setTimeout(() => {
                btnText.textContent = t.copyBtn;
            }, 2200);
        }

        updateDimensionVisibility();
        renderOptions();
        calculate();
    </script>
